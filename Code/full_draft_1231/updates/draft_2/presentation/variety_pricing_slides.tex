\section{Introduction}
\begin{frame}{Introduction}
\begin{block}{Research question}How do firms augment their optimal pricing strategy when love of variety is present?\end{block}
\begin{itemize}
\item Repeated consumption may reduce the appeal of a product.
\item Purchase histories can reveal when a consumer is ready to switch.
\item Retailers can leverage that information to optimally price according to the distribution of consumers.
  \begin{itemize}
    \item Dynamic price setting
    \item Targeting
  \end{itemize}
\end{itemize}
\note{Use the example of someone who likes berry yogurt but wants something different after repeated purchases. Distinguish a stable taste from fatigue.}
\end{frame}

\begin{frame}{Motivation}
\begin{block}{Motivation}
\begin{itemize}
\item \textbf{Pricing:} knowing how much consumers value variety helps firms tailor prices. In the model, this raises profit by about \$0.01 per consumer over two periods.
\item \textbf{Demand:} flavor preferences change little when fatigue is included; some fatigue parameters are weakly identified.
\end{itemize}
\end{block}
\begin{block}{Contribution}
\begin{itemize}
  \item Estimate distribution of LOV types 
  \begin{itemize}
    \item Lattin (1987), Chintagunta (1996), Dub\'e (2003)
  \end{itemize}
  \item Connect switching preferences and state dependence to the information a retailer uses for pricing
  \begin{itemize}
    \item Dub\'e and Misra (2023), Moorthy and Tehrani (2023), Karle and Reisinger (2024)
  \end{itemize}
\end{itemize}
\end{block}
\note{Explain upfront that the empirical demand model and toy supply model are separate exercises.

Position relative to attribute satiation in McAlister and Lattin, brand dynamics in Chintagunta, and personalized pricing in Dube and Misra. References are in the backup slides.}
\end{frame}

\section{Pricing Model}
\begin{frame}[label=pricing-setup]{Consumer problem: switching in both periods}
\small\setlength{\parskip}{3pt}
$r\in\{H,L\}$ is type; $t\in\{1,2\}$ is period; $\ell\in\{j,k\}$ indexes products.\\
$h_0=a\in\{j,k,0\}$ is the observed initial state; $h_1$ is the first choice. State $0$ means no prior product.
\[
U_{r\ell t}=\underbrace{\beta_\ell}_{\text{baseline utility}}
+\underbrace{\gamma_r\mathbf1\{h_{t-1}\ne0,\ell\ne h_{t-1}\}}_{\text{switching reward}}
+\underbrace{\alpha p_{\ell t}}_{\text{price effect}}+\varepsilon_{\ell t},\qquad U_{r0t}=\varepsilon_{0t}.
\]
\begin{itemize}
\item $\gamma_H>\gamma_L$: H values switching more; $\alpha<0$ is the common price coefficient.
\item $p_{\ell1}(a)$ and $p_{\ell2}(a,h_1)$ are freely chosen transaction prices.
  \item $\varepsilon$ are independent type-I extreme-value shocks; $V_{rlt}=U_{rlt}-\varepsilon_{lt}$ is deterministic utility.
\end{itemize}
\[
s_{r\ell t}=\Pr(h_t=\ell\mid r,\text{history})=
\frac{e^{V_{r\ell t}}}{1+\sum_{m\in\{j,k\}}e^{V_{rmt}}};
\quad s_{r0t}=1-s_{rjt}-s_{rkt}.
\]
\footnotesize Consumers maximize current utility. Initial states are equally likely and independent of type; a period-1 outside choice exits permanently.
\note{Both periods permit switching. Initial state 0 has no switching reward, while switching between j and k receives gamma. The firm observes initial state and subsequent choices. First purchases can be informative about type.}
\end{frame}

\begin{frame}[label=producer-problem]{Producer problem: induce, then harvest}
\small\setlength{\parskip}{3pt}
$P=\{p_{\ell1}(a),p_{\ell2}(a,h)\}$ is the full price menu; $c_\ell$ is unit cost.\\
$\mu_a=\Pr(h_0=a)=1/3$, $q=\Pr(H)$, and $\delta\in(0,1]$ discounts firm profit.
\[
\underbrace{w_r(a,h)}_{\text{terminal profit after first purchase }h}
=\sum_{\ell\in\{j,k\}}s_{r\ell2}(a,h)
\underbrace{[p_{\ell2}(a,h)-c_\ell]}_{\text{unit margin}},
\]
\[
\underbrace{\pi_r(P)}_{\text{value per arriving type-}r\text{ consumer}}
=\sum_{a\in\{j,k,0\}}\mu_a\sum_{h\in\{j,k\}}s_{rh1}(a)
\left[p_{h1}(a)-c_h+\delta w_r(a,h)\right].
\]
\[
P^*(q)\in\arg\max_P\{q\pi_H(P)+(1-q)\pi_L(P)\}.
\]
\begin{itemize}
\item Common menu: type is unobserved; prices depend on observed history. Observed type: optimize a separate menu for each $r$.
\item Commit before choices; account for type sorting into histories. Period-1 prices affect current earnings and future profit.
\end{itemize}
\note{There is no coupon restriction or imposed price bound. The outer expectation integrates over initial states; the inner expectation includes only surviving first purchases. Period-one outside choices earn no current or future profit.}
\end{frame}

\begin{frame}[label=optimal-price-conditions]{Optimal prices when type is known}
\small For type $r$, $c_\ell$ is unit cost, $\alpha<0$ is the price coefficient, and $s_{r0t}$ is the outside share. The initial state is $a$; the first purchase is $h$.
\par\medskip\textbf{Period 2: harvest}
\[
p^*_{r\ell2}(a,h)-c_\ell=-\frac{1}{\alpha s^*_{r02}(a,h)},\qquad \ell\in\{j,k\}.
\]
Terminal unit margins are equal across products at each history.

\par\medskip\textbf{Period 1: induce with continuation value}
\[
p^*_{r\ell1}(a)-c_\ell+\delta w_r^*(a,\ell)
=-\frac{1}{\alpha s^*_{r01}(a)},\qquad \ell\in\{j,k\}.
\]
$w_r^*(a,\ell)$ is optimal period-2 profit after buying $\ell$; $\delta$ is the firm discount factor.\\
Larger continuation value supports a lower current margin, holding the right-hand side fixed.

\footnotesize Interior conditions; shares and continuation values are evaluated at optimal prices.
\hfill\hyperlink{unobserved-price-conditions}{\beamerbutton{Unobserved type}}
\note{These known-type conditions use independent logit shocks and free prices. The first-period formula is the static logit markup condition applied to effective cost c minus delta w. The continuation value differs by the first product chosen. The linked appendix derives weighted conditions for unobserved types and accounts for selection into histories.}
\end{frame}

\begin{frame}[label=Simulation]{Free pricing: what changes?}

Baseline $(\gamma_H,\gamma_L)=(2,1)$; initial states $\mu_j=\mu_k=\mu_0=1/3$.
\end{block}
\begin{itemize}

\item For known type, interior terminal prices imply equal product margins at each history.
\item Measure induction by current profit sacrificed relative to the one-period optimum; measure harvesting by period-2 profit.
\end{itemize}

\section{Empirical exercise}
\begin{frame}[label=demand-model]{Demand model}
\[
U_{ijt}=\underbrace{\beta_0+\beta_i'X_{jt}}_{\text{baseline tastes}}
+\underbrace{\gamma_i\Xi_{ijt}}_{\text{history state}}
+\underbrace{\alpha p_{jt}}_{\text{price}}+\varepsilon_{ijt}.
\]

  \[
  \text{s.t.} \ 
  \Xi_{ijt} = \lambda\sum_{s=1}^8\delta^{t-s}\Xi_{ij,t-s} + (1-\lambda)\sum_{k\ne j}\rho_kX_{k,t-1}
\]
\begin{itemize}
\item Inside choices: berry, plain, other; outside utility normalized to zero.
\item $\Xi_{ijt}$ summarizes prior consumption and switching.
\item Negative fatigue coefficients reduce utility as the history index rises.
\end{itemize}
\note{The next slides define the full manuscript-style history index and show its update. Distinguish the intended fatigue motivation from the cross-flavor construction. Demand history coefficients and supply switching rewards have different sign conventions.}

\hyperlink{simulation-procedure}{\beamerbutton{Simulation procedure}}\quad
\hyperlink{simulation-results}{\beamerbutton{Simulation results}}\quad
\hyperlink{utility-specification}{\beamerbutton{Utility specification}}
\end{frame}

\begin{frame}{Identification}
\begin{itemize}
\item \textbf{Stable tastes:} persistent differences in flavor choices.
\item \textbf{Fatigue:} changes in choices associated with a household\textquotesingle s own history.
\item \textbf{Challenge:} heterogeneity, prices, and unobserved shocks can also generate switching.
\item \textbf{Limited history and coarse groups} make the distinction difficult.
\end{itemize}
\note{A switching pattern alone does not identify fatigue. Discuss initial conditions, serially correlated shocks, and the amount of within-household variation required.}
\end{frame}
\begin{frame}{Price control function}
\[
p_{jmt}=a+b\underbrace{\bar p_{j,-m,t}}_{\text{other-market price}}
+\tau_t+\eta_{jmt},\qquad
\hat\eta_{jmt}=p_{jmt}-\hat p_{jmt}.
\]
\begin{enumerate}
\item Match household choices to flavor-specific prices and purchase histories.
\item Regress local prices on other-market prices and time controls.
\item Include the fitted residual in utility to account for endogenous prices.
\end{enumerate}
\begin{block}{Required exclusion}Other-market prices must shift local prices without sharing unobserved local demand shocks, conditional on the controls.\end{block}
\note{This is the Petrin--Train control-function approach used in the draft. First-stage coefficients are written explicitly. The draft does not provide instrument strength diagnostics and is inconsistent about the number of other markets.}
\end{frame}

\section{Conclusion}
\begin{frame}{Conclusion}
\begin{itemize}
\item Restricting type information can reduce retailer profits in this model.
\item Type information changes optimal induction and harvesting prices.
\item Attribute fatigue lowers consumer utility
\item Difficult to disentangle preferences from love of variety
\end{itemize}
\note{Distinguish the illustrated cost of an information constraint from a causal estimate of an actual regulation. A policy conclusion needs consumer welfare, collection costs, and an explicit policy mapping.}
\end{frame}

\begin{frame}[noframenumbering,label=unobserved-price-conditions]{Optimal prices when type is unobserved}
\small\setlength{\parskip}{3pt}
Let $\omega_H=q$, $\omega_L=1-q$, $m_{\ell t}=p_{\ell t}-c_\ell$, and
\[
M_{r\ell}(a)=m_{\ell1}(a)+\delta w_r(a,\ell),\qquad
v_r(a)=\sum_{h\in\{j,k\}}s_{rh1}(a)M_{rh}(a).
\]
\textbf{Period 1: current margin plus continuation value}
\[
\sum_{r\in\{H,L\}}\omega_r s_{r\ell1}(a)
\left[1+\alpha\{M_{r\ell}(a)-v_r(a)\}\right]=0.
\]
\textbf{Period 2: harvest from the consumers reaching history $(a,h)$}
\[
\sum_{r\in\{H,L\}}\omega_r s_{rh1}(a)s_{r\ell2}(a,h)
\left[1+\alpha\{m_{\ell2}(a,h)-w_r(a,h)\}\right]=0.
\]
\footnotesize Conditions hold for each product and observed history. In period 2, $\omega_rs_{rh1}(a)$ accounts for type selection into the first purchase.
\par\medskip\hyperlink{optimal-price-conditions}{\beamerreturnbutton{Known-type conditions}}
\note{These are derivatives of the full committed-menu objective, with all other price coordinates held fixed. The period-two condition omits the positive common factor delta mu a; its type weights include first-period selection. The period-one condition includes continuation profits, which are held fixed in the price-coordinate derivative. Known-type formulas follow using the outside share at each history. No imposed price bounds or coupon constraints are used.}
\end{frame}

\begin{frame}[noframenumbering,label=supply-parameters]{Pricing exercise: parameters and timing}
\begin{itemize}
\item $\beta=(2.2,2.0)$, $c=(1.2,0.8)$, $\alpha=-1.5$, $\delta=0.95$.
\item Initial states $j,k,0$ have equal probabilities, independent of type; the firm observes the state.
\item Baseline $\gamma_H=2$, $\gamma_L=1$; sensitivity varies $\gamma_H$ from $1.5$ to $10$ in steps of $0.5$ with $q=0.5$.
\item Commit to free prices by observed history. Consumers choose myopically; first-period outside choices exit.
\item Three unconstrained optimizer starts per initial state, analytic gradients, and numerical stationarity checks.
\end{itemize}
\hyperlink{supply-results}{\beamerreturnbutton{Profits}}
\end{frame}

\begin{frame}[noframenumbering,label=pricing-details]{Free-price conditions: dynamic induction}
\small For known type, define $w_r(h)$ as terminal profit and $s_{r0t}$ as the outside share. At an interior optimum:
\[
p_{r\ell2}(h)-c_\ell=-\frac{1}{\alpha s_{r02}(h)},
\]
\[
p_{rh1}(a)-c_h+\delta w_r(a,h)=-\frac{1}{\alpha s_{r01}(a)}.
\]
\begin{itemize}
\item Terminal margins are equal across products for each known type and history.
\item Larger continuation value lowers the optimal first-period margin, all else equal.
\item With unobserved heterogeneous types, optimize the weighted full objective; known-type formulas cannot be applied to pooled shares.
\end{itemize}
\hyperlink{producer-problem}{\beamerreturnbutton{Producer problem}}
\end{frame}

\begin{frame}[noframenumbering,label=simulation-procedure]{Simulation procedure}
\small
\begin{tabular}{ll}\toprule
Simulation design & Assigned value\\\midrule
Replications / evaluation periods / burn-in & $1{,}000\;/\;100\;/\;5$\\
Product attributes / baseline taste & $X_j\in\{1,2,3,4,5\}\;;\;\beta=2$\\
Variety grid / choice shocks & $\gamma\in\{0,3,6,9\}\;;\;\varepsilon_{jt}\sim\mathrm{T1EV}$\\\bottomrule
\end{tabular}
\begin{enumerate}
\item For each $\gamma$, initialize consumption histories during burn-in.
\item At each evaluation date, combine baseline attribute utility with the history-dependent distance reward and a choice shock.
\item Record the selected product, update its consumption history, and repeat.
\item Summarize choice probabilities across the simulated runs.
\end{enumerate}
\textbf{Purpose:} show how variety changes choices in a controlled setting.
\note{The procedure summarizes the simulation described in the manuscript. Its distance-based mechanism differs from the empirical flavor-fatigue index. Simulation code is not re-estimated here.}

\begin{frame}[noframenumbering,label=utility-specification]{Utility specification}
\small
For $j\in\{B,P,O\}$ (berry, plain, other), let $D_j^f=\mathbf1\{j=f\}$:
\[
\begin{aligned}
U_{ijt}&=V_{ijt}+\varepsilon_{ijt},\\
V_{ijt}&=\beta_0+\beta_i^B D_j^B+\beta_i^P D_j^P\\
&\quad+\gamma_i^B D_j^B\Xi_{ijt}
+\gamma_i^P D_j^P\Xi_{ijt}
+\alpha p_{j,m(i),t}+\sigma\hat\eta_{j,m(i),t}.
\end{aligned}
\]
\begin{itemize}
\item Other flavor is the reference; outside deterministic utility is zero.
\item $\Xi_{ijt}$ summarizes own past consumption and alternative-flavor history.
\item History weight $\lambda=0.7$ and decay $\delta=0.9$ are fixed.
\item Random coefficients allow persistent household differences in tastes and fatigue.
\end{itemize}

The preceding slides define the manuscript-style history recursion using only past observed periods and explain its cross-flavor construction. The reference category normalizations match the reported tables.}

\begin{frame}[noframenumbering,label=panel-likelihood]{Panel likelihood}
\small
Conditional on household coefficients $b_i=(\beta_i^B,\beta_i^P,\gamma_i^B,\gamma_i^P)$:
\[
P_{ijt}(b_i)=\frac{e^{V_{ijt}(b_i)}}{1+\sum_{k\in\{B,P,O\}}e^{V_{ikt}(b_i)}},\qquad
P_{i0t}(b_i)=\frac{1}{1+\sum_k e^{V_{ikt}(b_i)}}.
\]
\[
L_i(b_i)=\prod_{t\in\mathcal T_i}\prod_{j\in\{0,B,P,O\}}
P_{ijt}(b_i)^{y_{ijt}},\qquad y_{ijt}=\mathbf1\{\text{observed choice is }j\}.
\]
\[
\boxed{\hat\theta=\arg\max_\theta\sum_i\log\!\left[\int L_i(b)\,dF(b;\theta)\right]}
\]
Simulation approximation: $\int L_i(b)dF(b;\theta)\approx R^{-1}\sum_{r=1}^R L_i(b_i^{(r)})$.

Maximize log-likelihood: $\ell(\theta)=\sum_{i,t,j}y_{ijt}\log P_{ijt}(\theta)$.
\note{Show that persistent household heterogeneity requires integrating the product over a household history, rather than multiplying separately integrated marginal probabilities. This is the full panel mixed-logit estimation target; the manuscript does not document the draw count or coefficient distribution, so neither is asserted here.}
